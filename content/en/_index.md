<script>
(function () {
  // pilih semua link nav yang menuju anchor di halaman ini
  const navLinks = document.querySelectorAll('a[href^="/#"], a[href^="#"]');

  function headerOffset() {
    // tinggi header + preheader (kalau ada & sedang terlihat)
    const header   = document.querySelector('header');
    const pre      = document.querySelector('.preheader');
    const h = header ? header.getBoundingClientRect().height : 0;

    // preheader hanya dihitung kalau masih terlihat (saat di top)
    const p = (pre && window.scrollY < 10) ? pre.getBoundingClientRect().height : 0;

    // tambahkan sedikit tweak supaya judul seksi “pas” di bawah header
    const tweak = (window.innerWidth < 768) ? 12 : 6; // naikkan/kurangi sesuai selera
    return h + p + tweak;
  }

  function scrollToHash(hash, pushstate=true) {
    const el = document.querySelector(hash);
    if (!el) return;

    const top = window.pageYOffset + el.getBoundingClientRect().top - headerOffset();
    window.scrollTo({ top, behavior: 'smooth' });

    // tutup menu mobile & update URL
    document.body.classList.remove('menushown');
    if (pushstate) history.pushState(null, '', hash);
  }

  // klik dari navbar
  navLinks.forEach(a => {
    a.addEventListener('click', function (e) {
      const href = this.getAttribute('href');
      const hash = href.startsWith('/#') ? href.slice(1) : href;
      if (hash.startsWith('#')) {
        e.preventDefault();
        scrollToHash(hash);
      }
    });
  });

  // jika halaman dibuka dengan hash, atur posisinya juga
  window.addEventListener('load', () => {
    if (location.hash) {
      // beri jeda agar header sticky siap
      setTimeout(() => scrollToHash(location.hash, false), 50);
    }
  });

  // kalau user ganti hash manual / via back-forward
  window.addEventListener('hashchange', () => {
    if (location.hash) scrollToHash(location.hash, false);
  });
})();
</script>
